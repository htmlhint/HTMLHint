        function initApp() {
            const stored = localStorage.getItem('meals_subscribers_data');
            subscribers = stored ? JSON.parse(stored) : generateDefaultData();
            saveData();
            renderStats();
            renderGrid();
        }

        function generateDefaultData() {
            return initialSubscribersList.map((item, index) => ({
                id: index + 1,
                page: item.page,
                sub_id: item.sub_id,
                name: item.name,
                phone: "",
                package: "اشتراك شهري",
                start_date: "",
                total_meals: 30,
                consumed: 0,
                remaining: 30
            }));
        }

        function saveData() {
            localStorage.setItem('meals_subscribers_data', JSON.stringify(subscribers));
            renderStats();
        }

        function renderStats() {
            document.getElementById('stat-total-subscribers').innerText = subscribers.length;
            document.getElementById('stat-total-meals').innerText = subscribers.reduce((a, c) => a + (parseInt(c.total_meals) || 0), 0);
            document.getElementById('stat-consumed-meals').innerText = subscribers.reduce((a, c) => a + (parseInt(c.consumed) || 0), 0);
            document.getElementById('stat-remaining-meals').innerText = subscribers.reduce((a, c) => a + (parseInt(c.remaining) || 0), 0);
        }

        function getBadgeInfo(r) {
            if (r > 10) return { class: 'badge-green', text: `${r} وجبة متبقية` };
            if (r > 0) return { class: 'badge-orange', text: `${r} وجبات متبقية` };
            return { class: 'badge-red', text: 'انتهت الوجبات' };
        }

        function renderGrid(query = '') {
            const grid = document.getElementById('cardsGrid');
            grid.innerHTML = '';
            const filtered = subscribers.filter(s => {
                const q = query.trim().toLowerCase();
                if (!q) return true;
                return s.name.toLowerCase().includes(q) || s.sub_id.toString().includes(q) || s.page.toString().includes(q);
            });

            if (filtered.length === 0) {
                grid.innerHTML = '<div class="no-results">لم يتم العثور على نتائج.</div>';
                return;
            }

            filtered.forEach(sub => {
                const badge = getBadgeInfo(sub.remaining);
                const card = document.createElement('div');
                card.className = 'subscriber-card';
                card.onclick = () => openModal(sub.id);
                card.innerHTML = `
                    <div class="card-header">
                        <span class="card-name">${sub.name}</span>
                        <span class="badge ${badge.class}">${badge.text}</span>
                    </div>
                    <div class="card-info">
                        <div class="info-row"><span>رقم الاشتراك:</span><span class="info-value">#${sub.sub_id}</span></div>
                        <div class="info-row"><span>رقم الصفحة:</span><span class="info-value">${sub.page}</span></div>
                    </div>
                    <div class="card-footer">
                        <span>المستهلك: <strong>${sub.consumed}</strong> / ${sub.total_meals}</span>
                        <button class="btn btn-primary" style="padding: 4px 10px; font-size: 12px;">التفاصيل</button>
                    </div>
                `;
                grid.appendChild(card);
            });
        }

        function handleSearch() {
            renderGrid(document.getElementById('searchInput').value);
        }

        function openModal(id) {
            currentSubscriberId = id;
            isEditing = false;
            const sub = subscribers.find(s => s.id === id);
            if (sub) {
                document.getElementById('modalTitle').innerText = `تفاصيل: ${sub.name}`;
                renderModalBody(sub);
                document.getElementById('subscriberModal').classList.add('active');
            }
        }

        function closeModal() {
            document.getElementById('subscriberModal').classList.remove('active');
        }
